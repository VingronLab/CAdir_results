piechart <- function(perc, title) {
  pie(
    x = c(perc, 1 - perc),
    labels = c("Markers", "Other"),
    col = rev(CAdir::mpimg_pal()(2))
  )
  title(main = title, cex.main = 1.2, line = 0.1)
}

rnks <- cabic@gene_ranks
markers <- scoreMarkers(sce, sce$cadir)

mat <- matrix(NA, nrow = length(rnks), ncol = length(rnks))
rownames(mat) <- names(rnks)
colnames(mat) <- names(rnks)

shared_genes <- as_tibble(mat)
perc_shared <- mat

for (i in seq_len(length(names(rnks)))) {
  c <- names(rnks)[i]
  cadir_genes <- rnks[[c]]$Rowname
  ngenes <- length(cadir_genes)

  for (j in seq_len(length(names(rnks)))) {
    d <- names(rnks)[j]
    cls_markers <- markers[[d]]
    cls_markers$Gene <- rownames(cls_markers)
    cls_markers <- cls_markers[cls_markers$mean.AUC >= 0.5, ]

    cls_markers <- cls_markers[
      order(cls_markers$mean.AUC, decreasing = TRUE),
    ]
    degs <- cls_markers$Gene[seq_len(ngenes)]

    overlap <- intersect(cadir_genes, degs)

    shared_genes[[c]][j] <- list(overlap)
    perc_shared[j, i] <- length(overlap) / ngenes
  }
}
df <- as_tibble(perc_shared)
df$scoreMarkers <- colnames(df)

df <- df %>%
  pivot_longer(names_to = "CAdir", values_to = "overlap", cols = -scoreMarkers)

p_shared <- ggplot(df, aes(x = CAdir, y = scoreMarkers, fill = overlap)) +
  geom_tile() +
  coord_fixed() +
  CAdir:::scale_fill_gradient_mpimg(name = "orange") +
  theme_bw()
