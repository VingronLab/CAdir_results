rerun_bench <- FALSE
res_exists <- file.exists(file.path(resdir, "sim_res_list_dircl.rds"))

if (rerun_bench | (!res_exists)) {
  iters <- seq.int(0, 100, by = 5)
  iters[1] <- 1
  # iters <- seq.int(0, 10, by = 5)
  # iters[1] <- 1

  sim_res_list_dircl <- vector(
    mode = "list",
    length = length(sim_files)
  )
  options(rlib_message_verbosity = "quiet")

  for (i in seq_len(length(sim_files))) {
    iter_vec <- vector(mode = "numeric", length = length(iters))
    ari_vec <- vector(mode = "numeric", length = length(iters))

    f <- sim_files[i]
    fnm <- gsub("_filtered.rds", "", basename(f))
    fnm <- gsub("dePROB-", "", fnm)
    fnm <- gsub("defacLOC-", "", fnm)
    fnm <- gsub("defacSCALE-", "", fnm)
    fnm <- gsub(".rds", "", fnm)
    ds <- stringr::str_split(f, "/")[[1]][5]
    fnm <- paste0(ds, "_", fnm)
    cat("\n", fnm)
    sce <- readRDS(f)

    dft <- df_sim %>%
      mutate(dataset_gsub = gsub("\\.", "_", sim)) %>%
      filter(dataset_gsub == fnm)

    if (dft$angle == 0) {
      cut_ang <- NULL
    } else {
      cut_ang <- dft$angle
    }

    set.seed(1234)
    sce_var <- scran::modelGeneVar(sce)
    sce_top <- scran::getTopHVGs(sce_var, n = dft$ntop, var.threshold = NULL)
    sce <- sce[sce_top, ]
    cnts <- as.matrix(logcounts(sce))

    ca <- cacomp(
      obj = cnts,
      princ_coords = 3,
      dims = dft$dims, #20
      top = nrow(cnts),
      residuals = "pearson",
      clip = TRUE
    )

    # Ensure that all iterations are initilized the same
    center_ids <- CAdir:::kmeanspp_init(
      points = ca@prin_coords_cols,
      k = dft$kdir
    )
    lines <- ca@prin_coords_cols[center_ids, ]

    for (j in seq_along(iters)) {
      n <- iters[j]
      cat("\n\titer:", n)

      set.seed(2358)
      cadir <- dirclust(
        points = ca@prin_coords_cols,
        k = dft$kdir,
        epochs = n,
        init = "kmeanspp",
        lines = lines,
        log = FALSE,
        convergence_thr = FALSE,
        max_iter = n
      )

      ari <- aricode::clustComp(cadir@cell_clusters, sce$Group)$ARI
      rm(cadir)
      ari_vec[j] <- ari
      iter_vec[j] <- n
    }

    sim_res_list_dircl[[i]] <- data.frame(
      dataset = fnm,
      n_iters = iter_vec,
      ARI = ari_vec
    )
  }
  saveRDS(
    sim_res_list_dircl,
    file = file.path(resdir, "sim_res_list_dircl.rds")
  )
} else {
  sim_res_list_dircl <- readRDS(file.path(resdir, "sim_res_list_dircl.rds"))
}
