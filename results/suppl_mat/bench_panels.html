pr1 <- ggplot(
  df1,
  aes(x = ndims, y = ARI, color = dataset, group = dataset)
) +
  geom_line(linewidth = 0.8) +
  geom_point(size = 1) +
  CAdir::scale_color_mpimg(name = "mpi_extend") +
  labs(
    title = "Nr. of dim. vs. ARI - real data",
    x = "dimensions",
    y = "ARI",
    color = "Dataset"
  ) +
  theme_bw() +
  theme(
    axis.text = element_text(size = 12, color = "black"),
    axis.title = element_text(size = 14),
    plot.title = element_text(size = 16),
    legend.text = element_text(size = 12),
    legend.title = element_text(size = 14)
  )
pr2 <- ggplot(
  df2,
  aes(x = aplq, y = ARI, color = dataset, group = dataset)
) +
  geom_line(linewidth = 0.8) +
  geom_point(size = 1) +
  CAdir::scale_color_mpimg(name = "mpi_extend") +
  labs(
    title = "APL quantile vs. ARI - real data",
    x = "APL cutoff quantile",
    y = "ARI",
    color = "Dataset"
  ) +
  theme_bw() +
  theme(
    axis.text = element_text(size = 12, color = "black"),
    axis.title = element_text(size = 14),
    plot.title = element_text(size = 16),
    legend.text = element_text(size = 12),
    legend.title = element_text(size = 14)
  )

  pr2 +
  # guides(fill = guide_legend(nrow = 3, byrow = TRUE)) +
  plot_layout(guides = "collect") +

  theme(
    plot.tag = element_text(size = 18, face = "bold"),
    legend.position = "bottom",
    legend.box = "vertical",
    legend.margin = margin()
  )
