qc_df <- perCellQCMetrics(sim)

reasons <- perCellQCFilters(
  qc_df,
  sum.field = "sum",
  detected.field = "detected",
  sub.fields = NULL
)
colData(sim) <- cbind(colData(sim), qc_df)
sim$discard <- reasons$discard
sim <- sim[, !reasons$discard]

# filter out genes expressed in less than 1% cells
bm <- counts(sim) > 0
ncell <- round(ncol(sim) * 0.01)
idx <- which((rowSums(bm) > ncell))
sim <- sim[idx, ]

# Normalize
clust_sim <- quickCluster(sim)
sim <- computeSumFactors(sim, cluster = clust_sim, min.mean = 0.1)
sim <- logNormCounts(sim)
sim <- runUMAP(sim)

sim_var <- scran::modelGeneVar(sim)
sim_top <- scran::getTopHVGs(sim_var, prop = 0.4)
sim <- sim[sim_top, ]
ca <- cacomp(

  dims = 50,
